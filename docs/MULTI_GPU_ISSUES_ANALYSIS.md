def _redistribute_workload(self, failed_gpu_id: int):

    # 鑾峰彇瀹屾暣鐨勫伐浣滆寖鍥翠俊鎭?

    with self._workers_lock:
        if failed_gpu_id not in self.workers:
            return
        
        failed_worker = self.workers[failed_gpu_id]
        worker_stats = failed_worker.get_stats()
    
    # 1. 璁＄畻鍓╀綑宸ヤ綔鑼冨洿

    total_range = worker_stats.get("key_range", {})
    keys_checked = worker_stats.get("keys_checked", 0)
    remaining_start = worker_stats.get("current_position", 0)
    
    # 2. 鑾峰彇鍋ュ悍宸ヤ綔鍣?

    healthy_gpus = [idx for idx in self.workers.keys() 
                   if idx != failed_gpu_id 
                   and not self.recovery_manager.is_gpu_failed(idx)]
    
    # 3. 鍒嗛厤鍓╀綑宸ヤ綔

    if remaining_start > 0:
        remaining_range = total_range - remaining_start
        chunk_size = remaining_range // len(healthy_gpus)
        
        for i, idx in enumerate(healthy_gpus):
            start = remaining_start + i * chunk_size
            end = start + chunk_size
            self.workers[idx].add_key_range(start, end)
